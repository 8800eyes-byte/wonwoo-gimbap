function doGet(e) { var action = e.parameter.action; var callback = e.parameter.callback; var result = {}; try { if (action === 'getInitialData') { result = getInitialData(e.parameter); } else if (action === 'submitOrder') { result = submitOrder(e.parameter); } else if (action === 'cancelOrder') { result = cancelOrder(e.parameter); } else if (action === 'adminLogin') { result = adminLogin(e.parameter); } else if (action === 'toggleOrderState') { result = toggleOrderState(e.parameter); } else if (action === 'transferData') { result = transferData(e.parameter); } else { result = { status: 'error', message: '알 수 없는 요청입니다.' }; } } catch (error) { result = { status: 'error', message: error.toString() }; } var jsonOutput = JSON.stringify(result); return ContentService.createTextOutput(callback + '(' + jsonOutput + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } function safeString(val) { return val ? String(val).replace(/\s+/g, '') : ""; } function safeDateString(val) { if (!val) return ""; if (val instanceof Date) return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd"); var str = String(val).trim(); if (str.match(/^\d{4}-\d{2}-\d{2}$/)) return str; var d = new Date(val); if (!isNaN(d.getTime())) return Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd"); return str; } // ⏰ 마감 여부 확인 (수동 마감 + 당일 오전 10시 자동 마감 + 임원 수동 해제 예외) function isDateClosed(dateStr) { if (!dateStr) return false; var props = PropertiesService.getScriptProperties(); // 1. 임원진이 10시 이후에 예외적으로 수동 '마감 해제'한 날짜인지 확인 var openedDates = JSON.parse(props.getProperty('openedDates') || "[]"); if (openedDates.indexOf(dateStr) > -1) { return false; } // 2. 임원진이 수동으로 '마감'한 날짜인지 확인 var closedDates = JSON.parse(props.getProperty('closedDates') || "[]"); if (closedDates.indexOf(dateStr) > -1) { return true; } // 3. 수령일 당일 오전 10시(10:00) 이후 자동 마감 체크 (한국 시간 기준) var tz = Session.getScriptTimeZone(); var now = new Date(); var todayStr = Utilities.formatDate(now, tz, "yyyy-MM-dd"); var currentHour = parseInt(Utilities.formatDate(now, tz, "H"), 10); if (dateStr === todayStr && currentHour >= 10) { return true; } if (dateStr < todayStr) { return true; } return false; } function getInitialData(params) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("명단") || ss.getSheetByName("월별_수령현황") || ss.getSheets()[0]; var data = sheet.getDataRange().getValues(); var members = []; if (data.length > 0) { var h = data[0]; var catIdx = h.indexOf("구분") > -1 ? h.indexOf("구분") : 0; var deptIdx = h.indexOf("학과") > -1 ? h.indexOf("학과") : 1; var nameIdx = h.indexOf("성명") > -1 ? h.indexOf("성명") : 3; for (var i = 1; i < data.length; i++) { if (data[i][catIdx] && data[i][nameIdx]) { members.push({ category: data[i][catIdx].toString().trim(), department: data[i][deptIdx].toString().trim(), name: data[i][nameIdx].toString().trim() }); } } } var targetDate = safeDateString(params.date); var isClosed = isDateClosed(targetDate); return { status: 'success', members: members, isClosed: isClosed }; } function submitOrder(params) { var targetDate = safeDateString(params.date); if (isDateClosed(targetDate)) { return { status: 'error', message: '신청이 마감되었습니다. (당일 오전 10시 자동 마감)' }; } var lock = LockService.getScriptLock(); try { lock.waitLock(10000); } catch (e) { return { status: 'error', message: '접속자가 많아 처리가 지연되었습니다. 다시 시도해 주세요.' }; } try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("신청현황"); SpreadsheetApp.flush(); var data = sheet.getDataRange().getValues(); var pCat = safeString(params.category); var pDept = safeString(params.department); var pName = safeString(params.name); var pDate = safeDateString(params.date); var canceledRowIndex = -1; for (var i = 1; i < data.length; i++) { var rowCat = safeString(data[i][0]); var rowDept = safeString(data[i][1]); var rowName = safeString(data[i][3]); var rowDate = safeDateString(data[i][4]); var rowStatus = safeString(data[i][5]); if (rowCat === pCat && rowDept === pDept && rowName === pName && rowDate === pDate) { if (rowStatus === "신청완료" || rowStatus === "수령완료") { return { status: 'error', message: '⚠️ 이미 김밥 신청이 완료된 원우님입니다. (중복 신청 불가)' }; } else if (rowStatus === "신청취소") { canceledRowIndex = i + 1; } } } var studentId = ""; var memberSheet = ss.getSheetByName("명단") || ss.getSheetByName("월별_수령현황"); if (memberSheet) { var memberData = memberSheet.getDataRange().getValues(); var h = memberData[0]; var cIdx = h.indexOf("구분") > -1 ? h.indexOf("구분") : 0; var dIdx = h.indexOf("학과") > -1 ? h.indexOf("학과") : 1; var idIdx = h.indexOf("학번") > -1 ? h.indexOf("학번") : 2; var nIdx = h.indexOf("성명") > -1 ? h.indexOf("성명") : 3; for (var j = 1; j < memberData.length; j++) { if (safeString(memberData[j][cIdx]) === pCat && safeString(memberData[j][dIdx]) === pDept && safeString(memberData[j][nIdx]) === pName) { studentId = memberData[j][idIdx] ? memberData[j][idIdx].toString().trim() : ""; break; } } } if (canceledRowIndex > -1) { if (studentId) { sheet.getRange(canceledRowIndex, 3).setValue(studentId); } sheet.getRange(canceledRowIndex, 6).setValue("신청완료"); SpreadsheetApp.flush(); return { status: 'success' }; } sheet.appendRow([params.category, params.department, studentId, params.name, pDate, "신청완료"]); SpreadsheetApp.flush(); return { status: 'success' }; } finally { lock.releaseLock(); } } function cancelOrder(params) { var targetDate = safeDateString(params.date); if (isDateClosed(targetDate)) { return { status: 'error', message: '신청이 마감되어 취소가 불가합니다. (당일 오전 10시 자동 마감)' }; } var lock = LockService.getScriptLock(); try { lock.waitLock(10000); } catch (e) { return { status: 'error', message: '잠시 후 다시 시도해 주세요.' }; } try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("신청현황"); var data = sheet.getDataRange().getValues(); var pCat = safeString(params.category); var pDept = safeString(params.department); var pName = safeString(params.name); var pDate = safeDateString(params.date); var isCanceled = false; for (var i = data.length - 1; i >= 1; i--) { if (safeString(data[i][0]) === pCat && safeString(data[i][1]) === pDept && safeString(data[i][3]) === pName && safeDateString(data[i][4]) === pDate && safeString(data[i][5]) === "신청완료") { sheet.getRange(i + 1, 6).setValue("신청취소"); isCanceled = true; } } if (isCanceled) { SpreadsheetApp.flush(); return { status: 'success' }; } return { status: 'error', message: '취소할 신청완료 내역을 찾지 못했습니다.' }; } finally { lock.releaseLock(); } } function adminLogin(params) { if (params.pw !== "1234") { return { status: 'error', message: '비밀번호 불일치' }; } var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("신청현황"); var data = sheet.getDataRange().getValues(); var applicants = []; var targetDate = safeDateString(params.date); for (var i = 1; i < data.length; i++) { if (safeDateString(data[i][4]) === targetDate && safeString(data[i][5]) === "신청완료") { applicants.push({ category: data[i][0].toString().trim(), dept: data[i][1].toString().trim(), studentId: data[i][2].toString().trim(), name: data[i][3].toString().trim(), date: safeDateString(data[i][4]) }); } } return { status: 'success', applicants: applicants, isClosed: isDateClosed(targetDate) }; } // 🔄 임원진 수동 마감 / 마감 해제 토글 (10시 자동 마감 예외 제어 포함) function toggleOrderState(params) { var props = PropertiesService.getScriptProperties(); var targetDate = safeDateString(params.date); var closedDates = JSON.parse(props.getProperty('closedDates') || "[]"); var openedDates = JSON.parse(props.getProperty('openedDates') || "[]"); var currentlyClosed = isDateClosed(targetDate); var newState; if (currentlyClosed) { // 현재 마감 상태(수동 마감 또는 10시 자동 마감) -> 수동으로 '마감 해제' 처리 var cIdx = closedDates.indexOf(targetDate); if (cIdx > -1) closedDates.splice(cIdx, 1); if (openedDates.indexOf(targetDate) === -1) openedDates.push(targetDate); newState = false; } else { // 현재 열려 있는 상태 -> 수동으로 '마감' 처리 var oIdx = openedDates.indexOf(targetDate); if (oIdx > -1) openedDates.splice(oIdx, 1); if (closedDates.indexOf(targetDate) === -1) closedDates.push(targetDate); newState = true; } props.setProperty('closedDates', JSON.stringify(closedDates)); props.setProperty('openedDates', JSON.stringify(openedDates)); return { status: 'success', isClosed: newState }; } function transferData(params) { var targetDate = safeDateString(params.date); var ss = SpreadsheetApp.getActiveSpreadsheet(); var orderSheet = ss.getSheetByName("신청현황"); var monthlySheet = ss.getSheetByName("월별_수령현황"); var rosterSheet = ss.getSheetByName("명단"); if (!monthlySheet) { monthlySheet = ss.insertSheet("월별_수령현황"); if (rosterSheet) { monthlySheet.getRange(1, 1, 1, 5).setValues([["순번", "구분", "학과", "학번", "성명"]]); monthlySheet.getRange(1, 1, 1, 5).setBackground("#ffff00"); } } var masterIdMap = {}; if (rosterSheet) { var rData = rosterSheet.getDataRange().getValues(); var rH = rData[0]; var rC = rH.indexOf("구분") > -1 ? rH.indexOf("구분") : 0; var rD = rH.indexOf("학과") > -1 ? rH.indexOf("학과") : 1; var rId = rH.indexOf("학번") > -1 ? rH.indexOf("학번") : 2; var rN = rH.indexOf("성명") > -1 ? rH.indexOf("성명") : 3; for (var k = 1; k < rData.length; k++) { var mCat = safeString(rData[k][rC]); var mDept = safeString(rData[k][rD]); var mName = safeString(rData[k][rN]); var mStuId = rData[k][rId] ? rData[k][rId].toString().trim() : ""; if (mCat && mDept && mName && mStuId) { masterIdMap[mCat + "|" + mDept + "|" + mName] = mStuId; } } if (monthlySheet) { var mData = monthlySheet.getDataRange().getValues(); var existingMap = {}; for (var m = 1; m < mData.length; m++) { var mKey = safeString(mData[m][1]) + "|" + safeString(mData[m][2]) + "|" + safeString(mData[m][4]); existingMap[mKey] = m + 1; } var newMembers = []; var maxRow = mData.length; for (var j = 1; j < rData.length; j++) { var cat = safeString(rData[j][rC]); var dept = safeString(rData[j][rD]); var name = safeString(rData[j][rN]); var stuId = rData[j][rId] ? rData[j][rId].toString().trim() : ""; if (cat && dept && name) { var key = cat + "|" + dept + "|" + name; if (existingMap[key]) { var mRow = existingMap[key]; var currentStuId = mData[mRow - 1][3] ? mData[mRow - 1][3].toString().trim() : ""; if (stuId && currentStuId !== stuId) { monthlySheet.getRange(mRow, 4).setValue(stuId); } } else { var newRow = [maxRow, rData[j][rC], rData[j][rD], stuId, rData[j][rN]]; var lastC = monthlySheet.getLastColumn(); for (var c = 5; c < lastC; c++) { newRow.push("-"); } newMembers.push(newRow); maxRow++; } } } if (newMembers.length > 0) { monthlySheet.getRange(monthlySheet.getLastRow() + 1, 1, newMembers.length, newMembers[0].length).setValues(newMembers); } } } var orderData = orderSheet.getDataRange().getValues(); var appliedUsers = {}; for (var i = 1; i < orderData.length; i++) { var oCat = safeString(orderData[i][0]); var oDept = safeString(orderData[i][1]); var oStuId = orderData[i][2] ? orderData[i][2].toString().trim() : ""; var oName = safeString(orderData[i][3]); var oKey = oCat + "|" + oDept + "|" + oName; if (masterIdMap[oKey] && oStuId !== masterIdMap[oKey]) { orderSheet.getRange(i + 1, 3).setValue(masterIdMap[oKey]); } if (safeDateString(orderData[i][4]) === targetDate && safeString(orderData[i][5]) === "신청완료") { appliedUsers[oKey] = true; } } var finalData = monthlySheet.getDataRange().getValues(); var lastCol = monthlySheet.getLastColumn(); if (lastCol < 5) lastCol = 5; var targetCol = lastCol + 1; for (var c = 5; c < lastCol; c++) { if (safeDateString(finalData[0][c]) === targetDate) { targetCol = c + 1; break; } } monthlySheet.getRange(1, targetCol).setValue("'" + targetDate); monthlySheet.getRange(1, targetCol).setBackground("#ffff00"); for (var r = 1; r < finalData.length; r++) { var cat = safeString(finalData[r][1]); var dept = safeString(finalData[r][2]); var name = safeString(finalData[r][4]); var key = cat + "|" + dept + "|" + name; if (appliedUsers[key]) { monthlySheet.getRange(r + 1, targetCol).setValue("신청완료"); } else { monthlySheet.getRange(r + 1, targetCol).setValue("-"); } } SpreadsheetApp.flush(); return { status: 'success' }; }
